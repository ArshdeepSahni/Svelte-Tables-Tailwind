Create `packages/core/vite.config.element.ts` — separate single-entry UMD build config for the Web Component. This is REQUIRED because Vite/Rolldown does not support UMD with multiple entry points (Critical Finding 1 from research). The existing vite.config.ts (multi-entry ESM+CJS) is untouched.

```typescript
// packages/core/vite.config.element.ts
import { defineConfig } from 'vite';
import dts from 'vite-plugin-dts';
import { resolve } from 'path';

export default defineConfig({
  build: {
    lib: {
      entry: resolve(__dirname, 'src/element.ts'),
      name: 'TcTable',
      fileName: 'element',
      // Single entry = UMD is supported
      formats: ['umd', 'es'],
    },
    outDir: 'dist',
    emptyOutDir: false, // preserve ESM+CJS output from main build
    rolldownOptions: {
      external: [], // Bundle lit into UMD for CDN self-contained usage
    },
    sourcemap: true,
    minify: true, // UMD is for CDN — minify for smaller download
  },
  plugins: [
    dts({
      include: ['src/element.ts'],
      outDir: 'dist',
      rollupTypes: false,
    }),
  ],
});
```

Create `packages/vue/vite.config.ts` — replaces tsup for Vue SFC processing. tsup cannot process .vue files (Critical Finding 2 from research). This config processes Table.vue and outputs ESM+CJS identical in structure to what tsup would have produced.

```typescript
// packages/vue/vite.config.ts
import { defineConfig } from 'vite';
import vue from '@vitejs/plugin-vue';
import dts from 'vite-plugin-dts';
import { resolve } from 'path';

export default defineConfig({
  plugins: [
    vue({
      // Tell Vue compiler that tc-table is a custom element — suppress unknown element warnings
      template: {
        compilerOptions: {
          isCustomElement: (tag) => tag.startsWith('tc-'),
        },
      },
    }),
    dts({
      include: ['src/**/*.ts', 'src/**/*.vue'],
      outDir: 'dist',
      rollupTypes: false,
    }),
  ],
  build: {
    lib: {
      entry: resolve(__dirname, 'src/index.ts'),
      formats: ['es', 'cjs'],
      fileName: 'index',
    },
    rolldownOptions: {
      external: ['vue', '@tablecraft/core'],
    },
    sourcemap: true,
    minify: false,
  },
});
```
