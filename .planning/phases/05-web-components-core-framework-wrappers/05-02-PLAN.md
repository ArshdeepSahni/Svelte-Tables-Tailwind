```typescript
import { LitElement, html, nothing } from 'lit';
import { isServer } from 'lit';
import { createTableEngine, resolveColumns, processRows } from './engine/index.js';
import type { Row, ResolvedColumn, ProcessedRow, CellDescriptor, ColumnOverrideShort, ColumnDefinition, ColumnInferenceMap } from './types.js';

export class TcTable extends LitElement {
  // CRITICAL: Light DOM — no shadow root. Shadow DOM is architecturally excluded (Phase 1 decision).
  override createRenderRoot(): HTMLElement | DocumentFragment {
    return this;
  }

  // CRITICAL: Use static properties (NOT @property() decorators).
  // Reason: tsconfig.base.json has useDefineForClassFields:true + target:ES2022.
  // @property() decorators fail with that combo. static properties = {} works unconditionally.
  // All properties must be initialized in constructor(), NOT as class field assignments.
  static override properties = {
    // Complex data — no attribute reflection (cannot serialize objects to HTML attributes)
    data:         { attribute: false },
    columns:      { attribute: false },
    selectedRows: { attribute: false },
    // Scalar attributes — reflected to/from HTML attributes
    rowHeight:  { type: Number,  attribute: 'row-height' },
    sortBy:     { type: String,  attribute: 'sort-by' },
    sortDir:    { type: String,  attribute: 'sort-dir' },
    bordered:   { type: Boolean },
    striped:    { type: Boolean },
    loading:    { type: Boolean },
    pageSize:   { type: Number,  attribute: 'page-size' },
    filterable: { type: Boolean },
    selectable: { type: Boolean },
    caption:    { type: String },
    // Internal reactive state (no attribute)
    _resolvedColumns: { state: true },
    _processedRows:   { state: true },
    _filterQuery:     { state: true },
  };

  // Public properties (initialized in constructor — NOT as class fields)
  declare data: Row[];
  declare columns: ColumnOverrideShort | ColumnDefinition[] | undefined;
  declare selectedRows: Row[];
  declare rowHeight: number;
  declare sortBy: string;
  declare sortDir: string;
  declare bordered: boolean;
  declare striped: boolean;
  declare loading: boolean;
  declare pageSize: number;
  declare filterable: boolean;
  declare selectable: boolean;
  declare caption: string;

  // Internal state
  declare private _resolvedColumns: ResolvedColumn[];
  declare private _processedRows: ProcessedRow[];
  declare private _filterQuery: string;
  private _engine: ReturnType<typeof createTableEngine> | null = null;
  private _inferenceMap: ColumnInferenceMap | null = null;

  constructor() {
    super();
    this.data = [];
    this.columns = undefined;
    this.selectedRows = [];
    this.rowHeight = 48;
    this.sortBy = '';
    this.sortDir = 'asc';
    this.bordered = false;
    this.striped = false;
    this.loading = false;
    this.pageSize = 50;
    this.filterable = false;
    this.selectable = false;
    this.caption = '';
    this._resolvedColumns = [];
    this._processedRows = [];
    this._filterQuery = '';
  }

  override willUpdate(changed: Map<string, unknown>): void {
    if (changed.has('data') || changed.has('columns')) {
      // Recompute immediately with text fallback (no inference result yet)
      this._resolvedColumns = resolveColumns(this.data, this.columns);
      this._processedRows = processRows(this.data, this._resolvedColumns);
      // Kick off inference in idle time — re-renders with typed columns when done
      this._scheduleInference();
    }
    if (changed.has('sortBy') || changed.has('sortDir')) {
      this._applySortFilter();
    }
    if (changed.has('_filterQuery')) {
      this._applySortFilter();
    }
  }

  private _scheduleInference(): void {
    if (isServer || typeof requestIdleCallback === 'undefined') return;
    if (!this.data.length) return;
    // Create engine for inference only — stateless computation
    const engine = createTableEngine({ data: this.data, columns: this.columns });
    engine.inferSchema(this.data, (inferenceMap: ColumnInferenceMap) => {
      this._inferenceMap = inferenceMap;
      // Re-resolve columns with inference result — triggers re-render via _resolvedColumns state
      this._resolvedColumns = resolveColumns(this.data, this.columns, inferenceMap);
      this._processedRows = processRows(this.data, this._resolvedColumns);
      this.requestUpdate();
    });
  }

  private _applySortFilter(): void {
    // Apply sort and filter to _processedRows
    let rows = processRows(this.data, this._resolvedColumns);
    if (this._filterQuery) {
      const q = this._filterQuery.toLowerCase();
      rows = rows.filter((row) =>
        Object.values(row.cells).some((cell) => {
          if ('value' in cell) return String(cell.value).toLowerCase().includes(q);
          if ('label' in cell) return String((cell as { label: string }).label).toLowerCase().includes(q);
          if ('formatted' in cell) return String((cell as { formatted: string }).formatted).toLowerCase().includes(q);
          return false;
        }),
      );
    }
    if (this.sortBy) {
      const dir = this.sortDir === 'desc' ? -1 : 1;
      rows = [...rows].sort((a, b) => {
        const aVal = a._original[this.sortBy];
        const bVal = b._original[this.sortBy];
        if (aVal == null) return 1 * dir;
        if (bVal == null) return -1 * dir;
        return (aVal < bVal ? -1 : aVal > bVal ? 1 : 0) * dir;
      });
    }
    this._processedRows = rows;
  }

  override render() {
    if (isServer) return nothing;
    return html`
      <table
        class="tc-table${this.bordered ? ' tc-table--bordered' : ''}${this.striped ? ' tc-table--striped' : ''}"
        role="grid"
        aria-busy=${this.loading ? 'true' : 'false'}
      >
        ${this.caption ? html`<caption>${this.caption}</caption>` : nothing}
        <thead>
          <tr role="row">
            ${this._resolvedColumns.map((col) => html`
              <th
                role="columnheader"
                scope="col"
                class="tc-th"
                aria-sort=${this.sortBy === col.key ? (this.sortDir === 'asc' ? 'ascending' : 'descending') : 'none'}
                @click=${() => this._handleSort(col)}
                style="cursor: pointer;"
              >
                ${col.label}
              </th>
            `)}
          </tr>
        </thead>
        <tbody @click=${this._handleBodyClick}>
          ${this._processedRows.map((row, rowIndex) => this._renderRow(row, rowIndex))}
        </tbody>
      </table>
    `;
  }

  private _renderRow(row: ProcessedRow, rowIndex: number) {
    return html`
      <tr role="row" class="tc-tr" data-row=${rowIndex} style="height: ${this.rowHeight}px;">
        ${this._resolvedColumns.map((col, colIndex) => html`
          <td role="gridcell" class="tc-td" data-col=${colIndex}>
            ${this._renderCell(row.cells[col.key] ?? { kind: 'text', value: '' })}
          </td>
        `)}
      </tr>
    `;
  }

  // Mirror of Cell.svelte — same CSS class names, same kind dispatch, same safety contract.
  // FOUND-07: Only 'html' kind uses innerHTML — descriptor.sanitized is DOMPurify-processed by the renderer.
  private _renderCell(descriptor: CellDescriptor) {
    switch (descriptor.kind) {
      case 'text':
        return html`<span>${descriptor.value}</span>`;
      case 'number':
        return html`<span class="tc-num">${descriptor.formatted}</span>`;
      case 'link':
        return html`<a
          class="tc-link"
          href=${descriptor.href}
          target=${descriptor.external ? '_blank' : nothing}
          rel=${descriptor.external ? 'noopener noreferrer' : nothing}
        >${descriptor.label}</a>`;
      case 'image':
        return html`<img src=${descriptor.src} alt=${descriptor.alt} class="tc-img" loading="lazy" />`;
      case 'badge':
        return html`<span class="tc-badge tc-badge--${descriptor.variant}">${descriptor.label}</span>`;
      case 'status':
        return html`<span class="tc-status tc-status--${descriptor.variant}" role="status">${descriptor.label}</span>`;
      case 'boolean':
        return html`<span class="tc-bool ${descriptor.value ? 'tc-bool--true' : 'tc-bool--false'}" aria-label=${descriptor.value ? 'Yes' : 'No'}>${descriptor.value ? '✓' : '✗'}</span>`;
      case 'code':
        return html`<code class="tc-code">${descriptor.value}</code>`;
      case 'html': {
        // sanitized is already DOMPurify-processed by the html renderer — safe to inject
        const span = document.createElement('span');
        span.innerHTML = descriptor.sanitized;
        return span;
      }
      default:
        return html`<span>${String((descriptor as { value: unknown }).value ?? '')}</span>`;
    }
  }

  private _handleSort(col: ResolvedColumn): void {
    if (this.sortBy === col.key) {
      this.sortDir = this.sortDir === 'asc' ? 'desc' : 'asc';
    } else {
      this.sortBy = col.key;
      this.sortDir = 'asc';
    }
    this.dispatchEvent(new CustomEvent('tc-sort-change', {
      detail: { key: this.sortBy, dir: this.sortDir },
      bubbles: true,
      composed: true,
    }));
  }

  private _handleBodyClick = (e: Event): void => {
    const target = e.target as HTMLElement;
    const td = target.closest('[data-col]') as HTMLElement | null;
    const tr = target.closest('[data-row]') as HTMLElement | null;
    if (!tr) return;
    const rowIndex = Number(tr.dataset['row']);
    const row = this._processedRows[rowIndex];
    if (!row) return;

    this.dispatchEvent(new CustomEvent('tc-row-click', {
      detail: { row: row._original, rowIndex },
      bubbles: true,
      composed: true,
    }));

    if (td) {
      const colIndex = Number(td.dataset['col']);
      const col = this._resolvedColumns[colIndex];
      if (col) {
        this.dispatchEvent(new CustomEvent('tc-cell-click', {
          detail: { row: row._original, column: col, value: row._original[col.key], rowIndex },
          bubbles: true,
          composed: true,
        }));
      }
    }
  };
}

// SSR guard — customElements is not available in Node.js / SvelteKit SSR
// This is the ONLY place customElements.define is called for tc-table
if (typeof customElements !== 'undefined') {
  customElements.define('tc-table', TcTable);
}
```
