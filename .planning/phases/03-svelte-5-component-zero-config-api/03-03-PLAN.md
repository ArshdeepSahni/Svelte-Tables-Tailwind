aria-sort values per WCAG spec:
- sortState.dir === 'asc' and key matches column → aria-sort="ascending"
- sortState.dir === 'desc' and key matches column → aria-sort="descending"
- column is sorted by a different key OR dir is null → aria-sort="none"
- column not sortable → no aria-sort attribute (aria-sort only valid on sortable columns)

Sort indicator text:
- ascending → '▲'
- descending → '▼'
- none/unsorted → '' (no indicator)

    ```typescript
      onselectall,
    }: Props = $props();
    ```

    Helper functions:
    ```typescript
    function getSortIndicator(colKey: string): string {
      if (sortState.key !== colKey) return '';
      if (sortState.dir === 'asc') return ' ▲';
      if (sortState.dir === 'desc') return ' ▼';
      return '';
    }

    function getAriaSortValue(colKey: string): 'ascending' | 'descending' | 'none' | undefined {
      if (!(columns.find((c) => c.key === colKey)?.sortable ?? true)) return undefined;
      if (sortState.key !== colKey) return 'none';
      if (sortState.dir === 'asc') return 'ascending';
      if (sortState.dir === 'desc') return 'descending';
      return 'none';
    }

    function handleHeaderClick(col: ResolvedColumn) {
      if (col.sortable === false) return;
      onsortchange?.(col.key);
    }
    ```

    Template:
    ```html
    <thead>
      <tr>
        {#if selectable}
          <th scope="col" class="tc-th tc-th--checkbox" role="columnheader">
            <Checkbox
              checked={allSelected}
              indeterminate={someSelected && !allSelected}
              oncheckedchange={(checked) => onselectall?.(checked)}
              aria-label="Select all rows"
            />
          </th>
        {/if}
        {#each columns as col (col.key)}
          <th
            scope="col"
            class="tc-th"
            class:tc-th--sortable={col.sortable !== false}
            class:tc-th--sorted={sortState.key === col.key && sortState.dir !== null}
            role="columnheader"
            aria-sort={getAriaSortValue(col.key)}
            onclick={() => handleHeaderClick(col)}
            onkeydown={(e) => { if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); handleHeaderClick(col); } }}
            tabindex={col.sortable !== false ? 0 : undefined}
          >
            {col.label}{getSortIndicator(col.key)}
          </th>
        {/each}
      </tr>
    </thead>
    ```

  <action>
    Create `packages/svelte/src/lib/Checkbox.svelte`.

    Critical implementation note: Svelte 5 has no `bind:indeterminate`. The `indeterminate` property must be set directly on the DOM element via `bind:this` + `$effect`. This is NOT a bug — it is the required pattern documented in the CONTEXT.md and research.

    Props:
    ```typescript
    interface Props {
      checked?: boolean;
      indeterminate?: boolean;
      disabled?: boolean;
      'aria-label'?: string;
      oncheckedchange?: (checked: boolean) => void;
    }

    let {
      checked = false,
      indeterminate = false,
      disabled = false,
      'aria-label': ariaLabel,
      oncheckedchange,
    }: Props = $props();
    ```

    State and ref:
    ```typescript
    let inputEl: HTMLInputElement | undefined = $state();
    ```

    Effect to set indeterminate DOM property (cannot be done via attribute):
    ```typescript
    $effect(() => {
      if (inputEl) {
        inputEl.indeterminate = indeterminate;
      }
    });
    ```

    Handler:
    ```typescript
    function handleChange(e: Event) {
      const target = e.target as HTMLInputElement;
      oncheckedchange?.(target.checked);
    }
    ```

    Template:
    ```html
    <input
      type="checkbox"
      class="tc-checkbox"
      bind:this={inputEl}
      {checked}
      {disabled}
      aria-label={ariaLabel}
      onchange={handleChange}
    />
    ```

    Style block: `.tc-checkbox` — cursor: pointer, width: 1rem, height: 1rem, accent-color: var(--tc-color-primary, #2563eb). When `disabled`, cursor: not-allowed, opacity: 0.5.

    Do NOT use `bind:checked={checked}` — that creates a two-way binding from inside the component that conflicts with the parent's `checked` prop. Use the one-way `{checked}` attribute with the `onchange` handler to communicate upward via the callback prop.
