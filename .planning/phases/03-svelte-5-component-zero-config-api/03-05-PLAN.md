Two-phase inference pattern (verified in research):
```typescript
$effect(() => {
  const engine = createTableEngine({ data: effectiveData, columns: effectiveColumns });

  // Phase 1: immediate text columns
  const textCols = engine.resolveColumns(effectiveData, effectiveColumns ?? undefined);
  resolvedColumns = textCols;
  processedRows = engine.processRows(effectiveData, textCols);

  // Phase 2: async typed columns via requestIdleCallback
  const schedule = typeof requestIdleCallback !== 'undefined' ? requestIdleCallback : (cb: () => void) => setTimeout(cb, 1);
  const cancel = typeof cancelIdleCallback !== 'undefined' ? cancelIdleCallback : clearTimeout;
  const idleId = schedule(() => {
    const schema = engine.inferSchema(effectiveData);
    const typedCols = engine.resolveColumns(effectiveData, effectiveColumns ?? undefined, schema);
    resolvedColumns = typedCols;
    processedRows = engine.processRows(effectiveData, typedCols);
  });

  return () => cancel(idleId as number);
});
```

Sort pipeline:
```typescript
const filteredRows = $derived.by(() => {
  if (!filterQuery.trim()) return processedRows;
  const q = filterQuery.toLowerCase();
  return processedRows.filter((row) =>
    Object.values(row.cells).some((cell) => {
      const v = 'value' in cell ? String((cell as {value: unknown}).value) :
                'label' in cell ? (cell as {label: string}).label :
                'formatted' in cell ? (cell as {formatted: string}).formatted : '';
      return v.toLowerCase().includes(q);
    })
  );
});

const sortedRows = $derived.by(() => {
  if (!sortState.key || !sortState.dir) return filteredRows;
  const col = resolvedColumns.find((c) => c.key === sortState.key);
  if (!col) return filteredRows;
  return [...filteredRows].sort((a, b) => {
    const av = a._original[sortState.key!];
    const bv = b._original[sortState.key!];
    const dir = sortState.dir === 'asc' ? 1 : -1;
    if (av == null) return dir;
    if (bv == null) return -dir;
    if (typeof av === 'number' && typeof bv === 'number') return (av - bv) * dir;
    return String(av).localeCompare(String(bv)) * dir;
  });
});

const totalPages = $derived(Math.ceil(filteredRows.length / pageSize));
const visibleRows = $derived(sortedRows.slice((currentPage - 1) * pageSize, currentPage * pageSize));
```
