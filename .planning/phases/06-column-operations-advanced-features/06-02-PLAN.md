    **Create packages/svelte/src/lib/AggregateRow.svelte:**
    ```svelte
    <script lang="ts">
      import type { ResolvedColumn, ProcessedRow } from '@tablecraft/core';
      import type { AggregateFunction } from '@tablecraft/core';

      interface Props {
        columns: ResolvedColumn[];
        rows: ProcessedRow[];
        selectable?: boolean;
      }

      let { columns, rows, selectable = false }: Props = $props();

      function computeAggregate(fn: AggregateFunction, values: unknown[]): string {
        if (typeof fn === 'function') return fn(values);
        const nums = values.map(Number).filter(isFinite);
        switch (fn) {
          case 'sum': return nums.reduce((a, b) => a + b, 0).toLocaleString();
          case 'avg': {
            if (nums.length === 0) return '—';
            const avg = nums.reduce((a, b) => a + b, 0) / nums.length;
            return avg.toLocaleString(undefined, { maximumFractionDigits: 2 });
          }
          case 'count': return String(values.length);
          case 'min': return nums.length ? Math.min(...nums).toLocaleString() : '—';
          case 'max': return nums.length ? Math.max(...nums).toLocaleString() : '—';
          default: return '—';
        }
      }
    </script>

    <tfoot>
      <tr class="tc-aggregate-row">
        {#if selectable}
          <td class="tc-td tc-aggregate tc-aggregate--label">Total</td>
        {/if}
        {#each columns as col (col.key)}
          <td class="tc-td tc-aggregate">
            {#if col.aggregate}
              {@const values = rows.map((r) => r._original[col.key])}
              {computeAggregate(col.aggregate, values)}
            {/if}
          </td>
        {/each}
      </tr>
    </tfoot>
    ```
