    ```typescript
        const ai = columnOrder.indexOf(a.key);
        const bi = columnOrder.indexOf(b.key);
    ```

    ```typescript
    function handleColReorder(fromKey: string, toKey: string) {
      if (fromKey === toKey) return;
      const newOrder = [...columnOrder];
      const fromIdx = newOrder.indexOf(fromKey);
      const toIdx = newOrder.indexOf(toKey);
      if (fromIdx === -1 || toIdx === -1) return;
      newOrder.splice(fromIdx, 1);
      newOrder.splice(toIdx, 0, fromKey);
      columnOrder = newOrder;
    }
    ```

    **TableHeader.svelte — add drag handlers:**
    ```typescript
    function handleDragStart(e: DragEvent, col: ResolvedColumn, th: HTMLElement) {
      // Prevent DnD when user is clicking on the resize handle
      const target = e.target as HTMLElement;
      if (target.classList.contains('tc-resize-handle')) {
        e.preventDefault();
        return;
      }
      draggingKey = col.key;
      e.dataTransfer!.effectAllowed = 'move';
      e.dataTransfer!.setData('text/plain', col.key);
      th.style.opacity = '0.5';
    }

    function handleDragEnd(e: DragEvent, th: HTMLElement) {
      draggingKey = null;
      dragoverKey = null;
      th.style.opacity = '';
    }

    function handleDragOver(e: DragEvent, col: ResolvedColumn) {
      if (isColumnVirtualizing) return; // reorder disabled with column virtualization
      e.preventDefault(); // CRITICAL: required for ondrop to fire
      e.dataTransfer!.dropEffect = 'move';
      dragoverKey = col.key;
    }

    function handleDrop(e: DragEvent, col: ResolvedColumn) {
      e.preventDefault();
      dragoverKey = null;
      const fromKey = e.dataTransfer!.getData('text/plain');
      if (fromKey && fromKey !== col.key) {
        oncolreorder?.(fromKey, col.key);
      }
    }

    function handleDragLeave() {
      dragoverKey = null;
    }
    ```

    **Add computePinnedOffset helper:**
    ```typescript
    function computePinnedOffset(targetCol: ResolvedColumn, side: 'left' | 'right'): number {
      const pinnedCols = columns.filter((c) => c.pinned === side);
      const idx = pinnedCols.findIndex((c) => c.key === targetCol.key);
      let offset = 0;
      for (let i = 0; i < idx; i++) {
        offset += colWidths.get(pinnedCols[i].key) ?? pinnedCols[i].minWidth ?? 150;
      }
      return offset;
    }
    ```
